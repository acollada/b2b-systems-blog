# Consumo de APIs REST externas mediante HttpClient y autenticación OAuth2  

## 1. Requisitos previos y contexto técnico  

| Tema | Descripción | Herramientas / Librerías recomendadas |
|------|-------------|----------------------------------------|
| **.NET** | El artículo se centra en .NET 6/7, pero los conceptos aplican a cualquier versión moderna de .NET. | `dotnet-sdk-7.0` |
| **HttpClient** | Clase de referencia para realizar peticiones HTTP. | `System.Net.Http` |
| **OAuth2** | Protocolo de autorización que permite a una aplicación acceder a recursos protegidos con un token de acceso. | `IdentityModel` (para simplificar el flujo de token) |
| **JSON** | Formato de intercambio de datos. | `System.Text.Json` o `Newtonsoft.Json` |
| **DI (Inyección de Dependencias)** | Es recomendable registrar `HttpClient` y los servicios de token en el contenedor de DI. | `Microsoft.Extensions.DependencyInjection` |
| **Configuración** | Se necesita un archivo `appsettings.json` con los datos de la API (URL, client_id, client_secret, scopes, etc.). | `Microsoft.Extensions.Configuration` |

### Flujo de OAuth2 recomendado

- **Client Credentials** (para APIs que no requieren autorización del usuario).  
- **Authorization Code** (si la API requiere acceso a recursos de un usuario).  
- En este artículo nos centraremos en **Client Credentials**, que es el más sencillo y el más usado en micro‑servicios.

### Por qué usar `HttpClientFactory`

- Evita el **“socket exhaustion”** (exhaustión de sockets) al crear y destruir instancias de `HttpClient`.  
- Permite la **configuración centralizada** y la **caché de DNS**.  
- Se integra con **policies de resiliencia** (retry, circuit breaker) mediante Polly.

## 2. Paso a paso de implementación con listas numeradas  

> **Nota**: Todo el código está pensado para una aplicación de consola o ASP.NET Core. Si usas otro tipo de proyecto, adapta los pasos en consecuencia.

### 2.1. Crear la configuración en `appsettings.json`

```json
{
  "ApiSettings": {
    "BaseUrl": "https://api.ejemplo.com",
    "TokenEndpoint": "https://auth.ejemplo.com/connect/token",
    "ClientId": "mi-client-id",
    "ClientSecret": "mi-client-secret",
    "Scope": "api.read"
  }
}
```

### 2.2. Definir modelos de configuración

```csharp
public class ApiSettings
{
    public string BaseUrl { get; set; } = string.Empty;
    public string TokenEndpoint { get; set; } = string.Empty;
    public string ClientId { get; set; } = string.Empty;
    public string ClientSecret { get; set; } = string.Empty;
    public string Scope { get; set; } = string.Empty;
}
```

### 2.3. Registrar servicios en `Program.cs`

```csharp
var builder = WebApplication.CreateBuilder(args);

// 1) Cargar configuración
builder.Services.Configure<ApiSettings>(builder.Configuration.GetSection("ApiSettings"));

// 2) Registrar HttpClient con Polly (retry + circuit breaker)
builder.Services.AddHttpClient("ApiClient")
    .ConfigureHttpClient((sp, client) =>
    {
        var settings = sp.GetRequiredService<IOptions<ApiSettings>>().Value;
        client.BaseAddress = new Uri(settings.BaseUrl);
        client.DefaultRequestHeaders.Accept.Add(new MediaTypeWithQualityHeaderValue("application/json"));
    })
    .AddPolicyHandler(Policy<HttpResponseMessage>
        .Handle<HttpRequestException>()
        .OrResult(msg => !msg.IsSuccessStatusCode)
        .WaitAndRetryAsync(3, retryAttempt => TimeSpan.FromSeconds(Math.Pow(2, retryAttempt))));

// 3) Registrar servicio de token (ver paso 4)
builder.Services.AddSingleton<ITokenService, TokenService>();

var app = builder.Build();
```

### 2.4. Implementar el servicio de obtención de token

```csharp
public interface ITokenService
{
    Task<string> GetAccessTokenAsync(CancellationToken cancellationToken = default);
}

public class TokenService : ITokenService
{
    private readonly ApiSettings _settings;
    private readonly IHttpClientFactory _clientFactory;
    private string? _cachedToken;
    private DateTimeOffset _expiresAt = DateTimeOffset.MinValue;
    private readonly SemaphoreSlim _semaphore = new(1, 1);

    public TokenService(IOptions<ApiSettings> settings, IHttpClientFactory clientFactory)
    {
        _settings = settings.Value;
        _clientFactory = clientFactory;
    }

    public async Task<string> GetAccessTokenAsync(CancellationToken cancellationToken = default)
    {
        // Si el token no ha expirado, devolvemos el cache
        if (!string.IsNullOrEmpty(_cachedToken) && DateTimeOffset.UtcNow < _expiresAt)
            return _cachedToken!;

        // Evitar que múltiples hilos soliciten el token simultáneamente
        await _semaphore.WaitAsync(cancellationToken);
        try
        {
            // Comprobamos de nuevo dentro del lock
            if (!string.IsNullOrEmpty(_cachedToken) && DateTimeOffset.UtcNow < _expiresAt)
                return _cachedToken!;

            var client = _clientFactory.CreateClient(); // HttpClient sin configuración específica
            var request = new HttpRequestMessage(HttpMethod.Post, _settings.TokenEndpoint);
            request.Content = new FormUrlEncodedContent(new[]
            {
                new KeyValuePair<string, string>("grant_type", "client_credentials"),
                new KeyValuePair<string, string>("client_id", _settings.ClientId),
                new KeyValuePair<string, string>("client_secret", _settings.ClientSecret),
                new KeyValuePair<string, string>("scope", _settings.Scope)
            });

            var response = await client.SendAsync(request, cancellationToken);
            response.EnsureSuccessStatusCode();

            var payload = await response.Content.ReadFromJsonAsync<TokenResponse>(cancellationToken: cancellationToken);

            _cachedToken = payload?.AccessToken ?? throw new InvalidOperationException("Token vacío");
            _expiresAt = DateTimeOffset.UtcNow.AddSeconds(payload?.ExpiresIn ?? 3600 - 60); // margen de 1 min

            return _cachedToken;
        }
        finally
        {
            _semaphore.Release();
        }
    }

    private sealed record TokenResponse(string AccessToken, int ExpiresIn);
}
```

### 2.5. Consumir la API con el token

```csharp
public interface IExternalApiService
{
    Task<T?> GetAsync<T>(string endpoint, CancellationToken ct = default);
    Task<T?> PostAsync<T>(string endpoint, object payload, CancellationToken ct = default);
}

public class ExternalApiService : IExternalApiService
{
    private readonly IHttpClientFactory _clientFactory;
    private readonly ITokenService _tokenService;

    public ExternalApiService(IHttpClientFactory clientFactory, ITokenService tokenService)
    {
        _clientFactory = clientFactory;
        _tokenService = tokenService;
    }

    private async Task<HttpClient> CreateClientAsync(CancellationToken ct = default)
    {
        var client = _clientFactory.CreateClient("ApiClient");
        var token = await _tokenService.GetAccessTokenAsync(ct);
        client.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", token);
        return client;
    }

    public async Task<T?> GetAsync<T>(string endpoint, CancellationToken ct = default)
    {
        var client = await CreateClientAsync(ct);
        var response = await client.GetAsync(endpoint, ct);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadFromJsonAsync<T>(cancellationToken: ct);
    }

    public async Task<T?> PostAsync<T>(string endpoint, object payload, CancellationToken ct = default)
    {
        var client = await CreateClientAsync(ct);
        var response = await client.PostAsJsonAsync(endpoint, payload, ct);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadFromJsonAsync<T>(cancellationToken: ct);
    }
}
```

public class TokenService : ITokenService
{
    // Variables de cache
    private string? _cachedToken;
    private DateTimeOffset _expiresAt = DateTimeOffset.MinValue;
    private readonly SemaphoreSlim _semaphore = new(1, 1);

    public async Task<string> GetAccessTokenAsync(CancellationToken cancellationToken = default)
    {
        if (!string.IsNullOrEmpty(_cachedToken) && DateTimeOffset.UtcNow < _expiresAt)
            return _cachedToken!;

        await _semaphore.WaitAsync(cancellationToken);
        try
        {
            if (!string.IsNullOrEmpty(_cachedToken) && DateTimeOffset.UtcNow < _expiresAt)
                return _cachedToken!;

            var client = _clientFactory.CreateClient(); // Cliente sin configuración
            var request = new HttpRequestMessage(HttpMethod.Post, _settings.TokenEndpoint)
            {
                Content = new FormUrlEncodedContent(new[]
                {
                    new KeyValuePair<string,string>("grant_type","client_credentials"),
                    new KeyValuePair<string,string>("client_id",_settings.ClientId),
                    new KeyValuePair<string,string>("client_secret",_settings.ClientSecret),
                    new KeyValuePair<string,string>("scope",_settings.Scope)
                })
            };

            var response = await client.SendAsync(request, cancellationToken);
            response.EnsureSuccessStatusCode();

            var payload = await response.Content.ReadFromJsonAsync<TokenResponse>(cancellationToken: cancellationToken);

            _cachedToken = payload?.AccessToken ?? throw new InvalidOperationException("Token vacío");
            _expiresAt = DateTimeOffset.UtcNow.AddSeconds(payload?.ExpiresIn ?? 3600 - 60); // 1 min margen

            return _cachedToken;
        }
        finally
        {
            _semaphore.Release();
        }
    }

    private sealed record TokenResponse(string AccessToken, int ExpiresIn);
}

public class ExternalApiService : IExternalApiService
